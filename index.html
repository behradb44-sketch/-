async function sendMessage(){ if(isSending) return; const text = input.value.trim(); if(!text && !selectedFile) return; isSending = true; sendButton.disabled = true; if(welcome){ welcome.style.display = "none"; } let userContent = text; // ================================ // IMAGE // ================================ if( selectedFile && selectedFile.type.startsWith("image/") ){ try{ const dataURL = await fileToDataURL(selectedFile); userContent = [ { type: "text", text: text || "این تصویر را بررسی کن." }, { type: "image_url", image_url: { url: dataURL } } ]; }catch(error){ console.error(error); } } // ================================ // VIDEO // ================================ else if( selectedFile && selectedFile.type.startsWith("video/") ){ userContent = text ? `${text}\n\n[ویدئو پیوست شد: ${selectedFile.name}]` : `[ویدئو پیوست شد: ${selectedFile.name}]`; } // ================================ // USER MESSAGE // ================================ const userMessage = { role: "user", content: typeof userContent === "string" ? userContent : ( text || `📎 ${selectedFile?.name || "فایل"}` ) }; messages.push(userMessage); renderMessage(userMessage); renderHistory(); scrollBottom(); input.value = ""; input.style.height = "auto"; clearFile(); // ================================ // THINKING WAVE // ================================ const thinkingRow = document.createElement("div"); thinkingRow.className = "message assistant"; thinkingRow.innerHTML = `
AI
در حال فکر کردن...
`; chatInner.appendChild( thinkingRow ); scrollBottom(); try{ // ================================ // API HISTORY // ================================ const apiMessages = messages .slice(-MAX_HISTORY) .map(message => ({ role: message.role, content: message.content })); // ================================ // STREAM REQUEST // ================================ const response = await fetch( "/api/chat/stream", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: apiMessages }) } ); if(!response.ok){ const raw = await response.text(); let message = `سرور پاسخ نامعتبر داد. ${response.status} HTTP`; try{ const data = JSON.parse(raw); if(data.error){ message = data.error; } }catch(e){} throw new Error(message); } if(!response.body){ throw new Error( "Streaming در مرورگر در دسترس نیست." ); } // موج را حذف کن thinkingRow.remove(); // ================================ // ASSISTANT MESSAGE // ================================ const assistantMessage = { role: "assistant", content: "" }; for( const event of events ){ const lines = event.split("\n"); for( const line of lines ){ if( !line.startsWith("data:") ){ continue; } const raw = line .slice(5) .trim(); if(!raw){ continue; } if(raw === "[DONE]"){ continue; } let data; try{ data = JSON.parse(raw); }catch(error){ console.warn( "Invalid SSE:", raw ); continue; } // ============================ // SEARCH // ============================ if( data.type === "search" ){ if(!searchBox){ searchBox = document.createElement( "div" ); searchBox.className = "search-status active"; searchBox.innerHTML = ` 🔎 در حال جستجو در وب... `; output.prepend( searchBox ); } const label = searchBox.querySelector( "span:last-child" ); let query = "در حال جستجو در وب..."; if( data.arguments ){ try{ const args = JSON.parse( data.arguments ); if(args.query){ query = `در حال جستجو برای «${args.query}»...`; } }catch(e){ // ممکن است arguments تکه‌تکه آمده باشد } } label.textContent = query; scrollBottom(); } // ============================ // REASONING // ============================ if( data.type === "reasoning" ){ // فعلاً متن reasoning را // نمایش نمی‌دهیم؛ موج باقی می‌ماند. scrollBottom(); } // ============================ // CONTENT // ============================ if( data.type === "content" ){ if(searchBox){ searchBox.remove(); searchBox = null; } fullText += data.text || ""; assistantMessage.content = fullText; output.innerHTML = renderMarkdown( fullText ); addCopyButtons( output ); scrollBottom(); } // ============================ // ERROR // ============================ if( data.type === "error" ){ throw new Error( data.message || "خطای Streaming" ); } } } } // ================================ // FINAL // ================================ if(!fullText.trim()){ fullText = "مدل پاسخ متنی برنگرداند."; output.textContent = fullText; } messages.push({ role: "assistant", content: fullText }); save(); renderHistory(); scrollBottom(); }catch(error){ console.error( "BEHRAD AI STREAM ERROR:", error ); thinkingRow.remove(); renderMessage({ role: "assistant", content: "❌ " + ( error.message || "خطای ناشناخته" ) }); scrollBottom(); }finally{ isSending = false; sendButton.disabled = false; input.focus(); } }
